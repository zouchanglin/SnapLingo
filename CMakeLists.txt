cmake_minimum_required(VERSION 3.24)
project(SnapLingo VERSION 0.3.0 LANGUAGES CXX)
if(DEFINED SNAPLINGO_MSVC_INCLUDE_PREFIX)
  # CMake may misdecode /showIncludes when only the Chinese MSVC language pack is installed.
  set(CMAKE_CL_SHOWINCLUDES_PREFIX "${SNAPLINGO_MSVC_INCLUDE_PREFIX}")
  set(CMAKE_CXX_CL_SHOWINCLUDES_PREFIX "${SNAPLINGO_MSVC_INCLUDE_PREFIX}")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets Network Concurrent Test)
add_library(snaplingo_core STATIC native/settings.cpp native/services.cpp native/capture.cpp)
target_include_directories(snaplingo_core PUBLIC native)
target_link_libraries(snaplingo_core PUBLIC Qt6::Widgets Qt6::Network Qt6::Concurrent windowsapp user32 gdi32)
target_compile_definitions(snaplingo_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
if(MSVC)
  add_compile_options(/utf-8 /W4 /permissive-)
  target_compile_options(snaplingo_core PRIVATE /utf-8 /W4 /permissive-)
endif()
qt_add_executable(SnapLingo WIN32 native/main.cpp native/window.cpp native/settingsdialog.cpp native/resources.qrc native/app.rc)
target_link_libraries(SnapLingo PRIVATE snaplingo_core)
set_target_properties(SnapLingo PROPERTIES WIN32_EXECUTABLE TRUE)
enable_testing()
qt_add_executable(native_tests native/tests.cpp native/settingsdialog.cpp)
set_target_properties(native_tests PROPERTIES WIN32_EXECUTABLE FALSE)
target_link_libraries(native_tests PRIVATE snaplingo_core Qt6::Test)
add_test(NAME native_regression COMMAND native_tests -platform offscreen)
