cmake_minimum_required(VERSION 3.24)
project(SnapLingo VERSION 0.3.0 LANGUAGES CXX)
if(DEFINED SNAPLINGO_MSVC_INCLUDE_PREFIX)
  # CMake may misdecode /showIncludes when only the Chinese MSVC language pack is installed.
  set(CMAKE_CL_SHOWINCLUDES_PREFIX "${SNAPLINGO_MSVC_INCLUDE_PREFIX}")
  set(CMAKE_CXX_CL_SHOWINCLUDES_PREFIX "${SNAPLINGO_MSVC_INCLUDE_PREFIX}")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets Network Concurrent Test)

# Keep the vendored Fluent-Qt library limited to its Widgets core.
set(FLUENT_QT_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(FLUENT_QT_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(FLUENT_QT_BUILD_GALLERY OFF CACHE BOOL "" FORCE)
set(FLUENT_QT_BUILD_PLUGINS OFF CACHE BOOL "" FORCE)
set(FLUENT_QT_BUILD_PARTICLES OFF CACHE BOOL "" FORCE)
set(FLUENT_QT_BUILD_SPATIAL OFF CACHE BOOL "" FORCE)
set(FLUENT_QT_BUILD_QML_BRIDGE OFF CACHE BOOL "" FORCE)
set(FLUENT_QT_BUILD_PYSIDE6_BINDINGS OFF CACHE BOOL "" FORCE)
set(FLUENT_QT_BUILD_WEBASSEMBLY_SUPPORT OFF CACHE BOOL "" FORCE)
set(FLUENT_QT_INSTALL OFF CACHE BOOL "" FORCE)
set(FLUENT_QT_ENABLE_WARNINGS OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/fluentqt EXCLUDE_FROM_ALL)

add_library(snaplingo_core STATIC settings.cpp services.cpp capture.cpp history.cpp)
target_include_directories(snaplingo_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(snaplingo_core PUBLIC Qt6::Widgets Qt6::Network Qt6::Concurrent FluentQt::FluentQt windowsapp user32 gdi32)
target_compile_definitions(snaplingo_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
if(MSVC)
  target_compile_options(snaplingo_core PRIVATE /utf-8 /W4 /permissive-)
endif()
qt_add_executable(SnapLingo WIN32 main.cpp window.cpp settingsdialog.cpp historydialog.cpp theme.cpp app_resources.qrc app.rc)
target_link_libraries(SnapLingo PRIVATE snaplingo_core FluentQt::FluentQt dwmapi)
set_target_properties(SnapLingo PROPERTIES WIN32_EXECUTABLE TRUE)
if(MSVC)
  target_compile_options(SnapLingo PRIVATE /utf-8 /W4 /permissive-)
endif()
enable_testing()
qt_add_executable(native_tests tests.cpp settingsdialog.cpp theme.cpp)
set_target_properties(native_tests PROPERTIES WIN32_EXECUTABLE FALSE)
target_link_libraries(native_tests PRIVATE snaplingo_core FluentQt::FluentQt Qt6::Test dwmapi)
if(MSVC)
  target_compile_options(native_tests PRIVATE /utf-8 /W4 /permissive-)
endif()
add_test(NAME native_regression COMMAND native_tests -platform offscreen)
